## Python 生成脚本

```python
# scripts/generate_excel_template.py
"""生成 120 门学科四级目录 Excel 空模板"""

from openpyxl import Workbook
from openpyxl.styles import PatternFill, Font, Alignment, Border, Side
from openpyxl.utils import get_column_letter

def create_master_template():
    wb = Workbook()
    
    # ====== Sheet 1: 课程总览 ======
    ws1 = wb.active
    ws1.title = "120门课程总览"
    
    headers = ["编号", "课程名称", "枝号·专业领域", "总学时", "学分",
               "教学层级", "前序依赖", "行业岗位覆盖"]
    
    header_fill = PatternFill(start_color="1E3A5F", fill_type="solid")
    header_font = Font(name="微软雅黑", bold=True, color="FFFFFF", size=11)
    thin_border = Border(
        left=Side(style="thin"), right=Side(style="thin"),
        top=Side(style="thin"), bottom=Side(style="thin")
    )
    
    for col, header in enumerate(headers, 1):
        cell = ws1.cell(row=1, column=col, value=header)
        cell.fill = header_fill
        cell.font = header_font
        cell.alignment = Alignment(horizontal="center", vertical="center")
        cell.border = thin_border
    
    # 设置列宽
    col_widths = [10, 22, 22, 10, 8, 22, 22, 30]
    for i, w in enumerate(col_widths, 1):
        ws1.column_dimensions[get_column_letter(i)].width = w
    
    # 120 行空行供专家填写
    for r in range(2, 122):
        for c in range(1, 9):
            cell = ws1.cell(row=r, column=c, value="")
            cell.border = thin_border
            cell.font = Font(name="微软雅黑", size=10)
    
    # ====== Sheet 2: 四级目录模板（单学科示例） ======
    ws2 = wb.create_sheet("四级目录模板(示例)")
    ws2.column_dimensions['A'].width = 60
    ws2.column_dimensions['B'].width = 40
    
    example_data = [
        ("学科编号：", "2-001"),
        ("课程名称：", "中药学"),
        ("", ""),
        ("第一篇  总论", ""),
        ("├── 第1章  中药学发展概况", ""),
        ("│   ├── 1.1  中药学的概念与学科定位", ""),
        ("│   │   └── 病证单位：中药的分子靶点网络定位", ""),
        ("│   │   └── 病证单位：中药的证候场论调控角色", ""),
    ]
    
    for r, (a, b) in enumerate(example_data, 1):
        ws2.cell(row=r, column=1, value=a).font = Font(name="微软雅黑", size=10)
        ws2.cell(row=r, column=2, value=b).font = Font(name="微软雅黑", size=10)
    
    # ====== Sheet 3: 学时汇总 ======
    ws3 = wb.create_sheet("学时分配汇总")
    # ... (填入上表数据)
    
    # ====== Sheet 4: 图例 ======
    ws4 = wb.create_sheet("图例与使用说明")
    # ... (填入编码/缩写说明)
    
    output = "~/tcm-knowledge-tree/docs/120-subjects-master.xlsx"
    wb.save(output)
    print(f"模板已生成：{output}")

if __name__ == "__main__":
    create_master_template()
```
